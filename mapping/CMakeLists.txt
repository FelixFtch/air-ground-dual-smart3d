cmake_minimum_required(VERSION 3.16)
project(air_ground_octomap LANGUAGES CXX)
find_package(octomap REQUIRED HINTS /opt/ros/humble/share/octomap)
find_package(nlohmann_json 3.2.0 REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(octomap_service octomap_service.cpp)
target_compile_features(octomap_service PRIVATE cxx_std_17)
target_compile_options(octomap_service PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(octomap_service PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(octomap_service PRIVATE ${OCTOMAP_LIBRARIES} nlohmann_json::nlohmann_json OpenSSL::Crypto)
install(TARGETS octomap_service RUNTIME DESTINATION bin)
